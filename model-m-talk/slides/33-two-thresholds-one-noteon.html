SwitchPhase::FirstActuated { tick: t1 } => {
    if delta >= SECOND_DELTA {
        let elapsed = tick.saturating_sub(t1);
        let velocity = if elapsed >= 80 { 1 } else {
            let v = 127u32
                .saturating_sub((elapsed * 126) / 80);
            (v + 1).min(127) as u8
        };
        Some(SwitchEvent::NoteOn { velocity })
    } else { None }
}
